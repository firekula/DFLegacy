cmake_minimum_required(VERSION 3.24)
project(DFLegacyLauncherStub LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "DFLegacy.LauncherStub requires Windows and MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /WX /utf-8 /permissive-)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
set(DFL_LAUNCHER_VERSION "1.0.13" CACHE STRING "Three-part launcher version")
if(NOT DFL_LAUNCHER_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
  message(FATAL_ERROR "Use a three-part launcher release version")
endif()
string(REPLACE "." "," DFL_VERSION_COMPONENTS "${DFL_LAUNCHER_VERSION}")
set(DFL_PRODUCT_VERSION "${DFL_LAUNCHER_VERSION}")
if(DFL_SOURCE_REVISION)
  string(LENGTH "${DFL_SOURCE_REVISION}" revision_length)
  if(NOT revision_length EQUAL 40 OR NOT DFL_SOURCE_REVISION MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "Set the source revision to a 40-character Git commit")
  endif()
  string(APPEND DFL_PRODUCT_VERSION "+${DFL_SOURCE_REVISION}")
endif()
set(DFL_ICON "${CMAKE_CURRENT_SOURCE_DIR}/../DFLegacy.Launcher/DFLegacy.Launcher.ico")
set(DFL_CLASSIC_SKIN "${CMAKE_CURRENT_SOURCE_DIR}/../DFLegacy.Launcher/Assets/Skin")
set(DFL_STARTUP_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../DFLegacy.Startup")
set(DFL_FILE_DESCRIPTION "DFLegacy Launcher")
set(DFL_EXECUTABLE_NAME "DFOLauncher.exe")
set(DFL_INTERNAL_NAME "DFOLauncher")
configure_file(DFOLauncher.rc.in "${CMAKE_CURRENT_BINARY_DIR}/DFOLauncher.rc" @ONLY)
set(DFL_PROMPT_RESOURCE "${CMAKE_CURRENT_BINARY_DIR}/prompt.rc")
configure_file(prompt.rc.in "${DFL_PROMPT_RESOURCE}" @ONLY)
set_source_files_properties("${DFL_PROMPT_RESOURCE}" PROPERTIES OBJECT_DEPENDS
    "${DFL_CLASSIC_SKIN}/classic-message-panel.png;${DFL_CLASSIC_SKIN}/classic-message-confirm0.png;${DFL_CLASSIC_SKIN}/classic-message-confirm2.png")
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/DFOLauncher.rc" PROPERTIES OBJECT_DEPENDS "${DFL_ICON}")
add_executable(DFOLauncher WIN32 main.cpp layout.cpp launch.cpp process.cpp journal.cpp migration.cpp shortcut.cpp ui.cpp single_instance.cpp "${CMAKE_CURRENT_BINARY_DIR}/DFOLauncher.rc" "${DFL_PROMPT_RESOURCE}")
target_include_directories(DFOLauncher PRIVATE "${DFL_STARTUP_SOURCE}")
target_link_libraries(DFOLauncher PRIVATE shell32 ole32 shlwapi gdiplus uuid)
include(CTest)
if(BUILD_TESTING)
  add_executable(fixture tests/fixture.cpp)
  add_executable(stub_tests tests/stub_tests.cpp layout.cpp launch.cpp process.cpp journal.cpp migration.cpp shortcut.cpp single_instance.cpp)
  target_link_libraries(stub_tests PRIVATE ole32 uuid)
  add_test(NAME stub_layout_and_migration COMMAND stub_tests "$<TARGET_FILE:fixture>")
endif()
