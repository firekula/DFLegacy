cmake_minimum_required(VERSION 3.24)
project(DFLegacyStartup LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "DFLegacy.Startup requires Windows and MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /WX /utf-8 /permissive-)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(bootstrap STATIC bootstrap.cpp)
target_include_directories(bootstrap PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(bootstrap PUBLIC bcrypt crypt32)
set(DFL_CLASSIC_SKIN "${CMAKE_CURRENT_SOURCE_DIR}/../DFLegacy.Launcher/Assets/Skin")
set(startup_prompt_skin "${CMAKE_CURRENT_BINARY_DIR}/startup_prompt.rc")
configure_file(startup_prompt.rc.in "${startup_prompt_skin}" @ONLY)
set_source_files_properties("${startup_prompt_skin}" PROPERTIES OBJECT_DEPENDS
    "${DFL_CLASSIC_SKIN}/classic-message-panel.png;${DFL_CLASSIC_SKIN}/classic-message-confirm0.png;${DFL_CLASSIC_SKIN}/classic-message-confirm1.png;${DFL_CLASSIC_SKIN}/classic-message-confirm2.png")
set(launcher_version_file "${CMAKE_CURRENT_SOURCE_DIR}/../DFLegacy.Launcher/LauncherVersion.txt")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${launcher_version_file}")
if(NOT DEFINED DFL_LAUNCHER_VERSION)
  file(STRINGS "${launcher_version_file}" DFL_LAUNCHER_VERSION LIMIT_COUNT 1)
endif()
if(NOT DFL_LAUNCHER_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
  message(FATAL_ERROR "Use a three-part launcher release version, without a v prefix")
endif()
string(REPLACE "." "," DFL_VERSION_COMPONENTS "${DFL_LAUNCHER_VERSION}")
set(DFL_PRODUCT_VERSION "${DFL_LAUNCHER_VERSION}")
if(DFL_SOURCE_REVISION)
  string(LENGTH "${DFL_SOURCE_REVISION}" revision_length)
  if(NOT revision_length EQUAL 40 OR NOT DFL_SOURCE_REVISION MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "Set the source revision to a 40-character Git commit")
  endif()
  string(APPEND DFL_PRODUCT_VERSION "+${DFL_SOURCE_REVISION}")
endif()
set(DFL_LAUNCHER_ICON "${CMAKE_CURRENT_SOURCE_DIR}/../DFLegacy.Launcher/DFLegacy.Launcher.ico")

function(configure_launcher_resource role description executable)
  set(DFL_FILE_DESCRIPTION "${description}")
  set(DFL_EXECUTABLE_NAME "${executable}")
  get_filename_component(DFL_INTERNAL_NAME "${executable}" NAME_WLE)
  if(NOT role STREQUAL "entry")
    set(DFL_MANAGED_MANIFEST_RESOURCE "1 24 \"${CMAKE_CURRENT_SOURCE_DIR}/launcher.manifest\"")
  endif()
  configure_file(launcher.rc.in "${CMAKE_CURRENT_BINARY_DIR}/${role}.rc" @ONLY)
  set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/${role}.rc" PROPERTIES OBJECT_DEPENDS "${DFL_LAUNCHER_ICON}")
endfunction()

configure_launcher_resource(entry "DFLegacy Launcher Startup" "DFLegacy.Launcher.exe")
configure_launcher_resource(app "DFLegacy Launcher" "DFLegacy.Launcher.App.exe")
configure_launcher_resource(updater "DFLegacy Client Updater" "DFLegacy.Updater.exe")
configure_launcher_resource(standalone "DFLegacy Chapter Two Launcher" "DFLegacy.Chapter2.Launcher.exe")
add_executable(DFLegacy.Launcher WIN32 main.cpp "${CMAKE_CURRENT_BINARY_DIR}/entry.rc" startup_prompt.cpp "${startup_prompt_skin}")
target_link_libraries(DFLegacy.Launcher PRIVATE bootstrap shell32 ole32 gdiplus)
include(CTest)
if(BUILD_TESTING)
  foreach(result IN ITEMS 0 1 2 3 7)
    add_executable(fixture_${result} tests/fixture.cpp)
    target_compile_definitions(fixture_${result} PRIVATE FIXTURE_RESULT=${result})
  endforeach()
  add_executable(bootstrap_tests tests/bootstrap_tests.cpp)
  target_link_libraries(bootstrap_tests PRIVATE bootstrap)
  add_test(NAME bootstrap_policy_and_process COMMAND bootstrap_tests
    "$<TARGET_FILE:fixture_0>" "$<TARGET_FILE:fixture_1>"
    "$<TARGET_FILE:fixture_2>" "$<TARGET_FILE:fixture_7>" "$<TARGET_FILE:fixture_3>"
    "$<TARGET_FILE:DFLegacy.Launcher>")
  set_tests_properties(bootstrap_policy_and_process PROPERTIES TIMEOUT 60)
endif()
