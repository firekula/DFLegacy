cmake_minimum_required(VERSION 3.24)
project(DFLegacyStartup LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "DFLegacy.Startup requires Windows and MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /WX /utf-8 /permissive-)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(bootstrap STATIC bootstrap.cpp)
target_include_directories(bootstrap PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(bootstrap PUBLIC bcrypt crypt32)
set(DFL_CLASSIC_SKIN "${CMAKE_CURRENT_SOURCE_DIR}/../DFLegacy.Launcher/Assets/Skin")
set(startup_prompt_skin "${CMAKE_CURRENT_BINARY_DIR}/startup_prompt.rc")
configure_file(startup_prompt.rc.in "${startup_prompt_skin}" @ONLY)
set_source_files_properties("${startup_prompt_skin}" PROPERTIES OBJECT_DEPENDS
    "${DFL_CLASSIC_SKIN}/classic-message-panel.png;${DFL_CLASSIC_SKIN}/classic-message-confirm0.png;${DFL_CLASSIC_SKIN}/classic-message-confirm2.png")
add_executable(DFLegacy.Launcher WIN32 main.cpp launcher.rc startup_prompt.cpp "${startup_prompt_skin}")
target_link_libraries(DFLegacy.Launcher PRIVATE bootstrap shell32 ole32 gdiplus)
include(CTest)
if(BUILD_TESTING)
  foreach(result IN ITEMS 0 1 2 3 7)
    add_executable(fixture_${result} tests/fixture.cpp)
    target_compile_definitions(fixture_${result} PRIVATE FIXTURE_RESULT=${result})
  endforeach()
  add_executable(bootstrap_tests tests/bootstrap_tests.cpp)
  target_link_libraries(bootstrap_tests PRIVATE bootstrap)
  add_test(NAME bootstrap_policy_and_process COMMAND bootstrap_tests
    "$<TARGET_FILE:fixture_0>" "$<TARGET_FILE:fixture_1>"
    "$<TARGET_FILE:fixture_2>" "$<TARGET_FILE:fixture_7>" "$<TARGET_FILE:fixture_3>"
    "$<TARGET_FILE:DFLegacy.Launcher>")
  set_tests_properties(bootstrap_policy_and_process PROPERTIES TIMEOUT 60)
endif()
